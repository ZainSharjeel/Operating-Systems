#include "types.h"
#include "stat.h"
#include "user.h"

int main(int argc, char *argv[]) {
  if(argc < 2) {
    printf(2, "Usage: syscall_counter <command>\n");
    exit();
  }

  // Fork a child process
  int pid = fork();
  if(pid < 0) {
    printf(2, "Fork failed\n");
    exit();
  }

  // In the child process, execute the command
  if(pid == 0) {
    exec(argv[1], &argv[1]);
    printf(2, "Exec failed\n");
    exit();
  }

  // In the parent process, wait for the child to finish
  wait();

  // Print the system call count for each system call
  for(int i = 0; i < 22; i++) {
    int count = getsyscallcount(i);  // New system call
    if(count > 0) {
      printf(1, "Syscall %d invoked %d times\n", i, count);
    }
  }

  exit();
}
